#!/bin/bash

EXPAND_AMOUNT_MIB=500

while true; do
    USAGE=$(df /data | awk 'NR==2 {gsub(/%/, "", $5); print $5}')
    VG_FREE_MIB=$(vgs -o vg_free --units m --noheadings --nosuffix vg0 | cut -d'.' -f1)

    if [ "$USAGE" -gt 90 ]; then
        if [ "$VG_FREE_MIB" -ge "$EXPAND_AMOUNT_MIB" ]; then
            lvextend -L +"${EXPAND_AMOUNT_MIB}M" /dev/vg0/data0
            resize2fs /dev/vg0/data0
            echo "Expanded /data by ${EXPAND_AMOUNT_MIB}M. Free space in VG: ${VG_FREE_MIB}MiB."
        else
            echo "WARNING: Insufficient space ($VG_FREE_MIB MiB free). ${EXPAND_AMOUNT_MIB}M required for expansion."
        fi
    fi
    
    sleep 2
done
